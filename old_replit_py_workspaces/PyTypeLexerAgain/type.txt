class Type:
	def __init__(self, name: str, scope: Dict[str, "Type"] = GLOBALTYPES):
		self.fullname = name
		self.name = name.split(".")[-1]
		self.__inherits__: List["Type"] = [self]
		self.__literals__: List[type] = []
		self.static = {}
		self.data = {}
		self = scope.setdefault(self.fullname, self)
	@property
	def Type(self) -> "Type":
		return self
	def inheritance(self, *type: "Type"):
		types = map(lambda i: (
			None if self.__inherits__.count(i) else i), type)
		self.__inherits__.extend(list(types))
	def literals(self, *type: "LiteralType"):
		types = map(lambda i: (
			None if self.__literals__.count(i) else i), type)
		self.__literals__.extend(list(types))
	def isliteral(self, literal: "LiteralType") -> bool:
		for typ in self.__inherits__:
			if self.__literals__.count(literal):
				return True
			return typ.isliteral(type)
		return False
	def inherits(self, type: "Type") -> bool:
		if type.fullname == "builtins.Any": return True
		for typ in self.__inherits__:
			if typ.fullname == type.fullname:
				return True
			if typ == self: continue
			return typ.inherits(type)
		return False
	def __repr__(self) -> str:
		return self.name

	"""
 	@Non-Mutating
  	Returns the value obtained from the data or default if None
 	"""
	def get(self, key, default = None)->Any:
		return self.data[key] if key in self.data else default
	"""
 	@Mutating
  	Inserts the value in the data at the key
 	"""
	def set(self, key, value):
		self.data[key] = value
	"""
 	@Mutating
  	Sets the value at key if empty and returns the value obtained
 	"""
	def attr(self, key, default)->Any:
		return self.data.setdefault(key, default)
